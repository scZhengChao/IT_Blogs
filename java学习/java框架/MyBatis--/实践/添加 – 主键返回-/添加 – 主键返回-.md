添加 – 主键返回

在mybatis的insert操作过程中，存在一种技术：主键回填 &#x20;

在向表中插入一行记录后，**会自动把当前行的主键列下的数据值获取**

![](image_rfk0zGwCaW.png)

```java title="sql-xml"
<?xml version="1.0" encoding="UTF-8" ?>
<!DOCTYPE mapper
        PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
        "http://mybatis.org/dtd/mybatis-3-mapper.dtd">

<mapper namespace="com.itheima.dao.OrderMapper">

    <!--添加订单-->
    <insert id="addOrder" useGeneratedKeys="true" keyProperty="id">
        insert into tb_order (payment, payment_type, status)
        values (#{payment},#{paymentType},#{status})
    </insert>

    <!--添加订单明细-->
    <insert id="addOrderItem">
        insert into tb_order_item (goods_name, goods_price, count, order_id)
        values (#{goodsName},#{goodsPrice},#{count},#{orderId})
    </insert>

</mapper>
```


返回添加数据的主键\<insert  useGeneratedKeys="true" keyProperty="id">

```java 
import com.itheima.dao.OrderMapper;
import com.itheima.pojo.Order;
import com.itheima.pojo.OrderItem;
import com.itheima.util.MybatisUtil;
import org.apache.ibatis.session.SqlSession;
import org.junit.Test;

public class OrderMapperTest {


    @Test
    public void testAddOrder(){
        SqlSession sqlSession = MybatisUtil.openSession();

        OrderMapper orderMapper = sqlSession.getMapper(OrderMapper.class);

        //订单对象
        Order order = new Order();
        order.setPayment(10000.0);
        order.setStatus(1);
        order.setPaymentType(2);//2表示支付宝
       // System.out.println(order);
        int count1 = orderMapper.addOrder( order);//count1是受影响行数。 主键回填：插入行的主键值填充到order对象的id属性中
       //  System.out.println(order);
        //订单项对象
        OrderItem orderItem = new OrderItem();
         orderItem.setOrderId( order.getId() );//从Order对象中获取主键值 
         orderItem.setGoodsName("小米");
        orderItem.setGoodsPrice(1000);
        orderItem.setCount("10");
        int count2 = orderMapper.addOrderItem(orderItem);


        if(count1 > 0 && count2>0){
           sqlSession.commit();
            System.out.println("订单生成完成!");
        }

        sqlSession.close();
    }
}
```
