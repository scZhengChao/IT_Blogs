# 七、HTML标签大全列表

以下是更全面的HTML标签列表（按字母顺序）：

```html 
<!-- 文档结构 -->
<!DOCTYPE>, <html>, <head>, <body>
<!-- 元信息 -->
<base>, <link>, <meta>, <style>, <title>
<!-- 内容分区 -->
<address>, <article>, <aside>, <footer>, <header>, <h1>-<h6>, <main>, <nav>, <section>
<!-- 文本内容 -->
<blockquote>, <dd>, <div>, <dl>, <dt>, <figcaption>, <figure>, <hr>, <li>, <ol>, <p>, <pre>, <ul>
<!-- 内联文本语义 -->
<a>, <abbr>, <b>, <bdi>, <bdo>, <br>, <cite>, <code>, <data>, <dfn>, <em>, <i>, <kbd>, <mark>, <q>, <rp>, <rt>, <ruby>, <s>, <samp>, <small>, <span>, <strong>, <sub>, <sup>, <time>, , <var>, <wbr>
<!-- 图片和多媒体 -->
<area>, <audio>, <img>, <map>, <track>, <video>
<!-- 内嵌内容 -->
<embed>, <iframe>, <object>, <param>, <picture>, <portal>, <source>
<!-- SVG和MathML -->
<svg>, <math>
<!-- 脚本 -->
<canvas>, <noscript>, <script>
<!-- 表格 -->
<caption>, <col>, <colgroup>, <table>, <tbody>, <td>, <tfoot>, <th>, <thead>, <tr>
<!-- 表单 -->
<button>, <datalist>, <fieldset>, <form>, <input>, <label>, <legend>, <meter>, <optgroup>, <option>, <output>, <progress>, <select>, <textarea>
<!-- 交互元素 -->
<details>, <dialog>, <menu>, <summary>
<!-- Web组件 -->
<slot>, <template>
```
